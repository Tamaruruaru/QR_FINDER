function mostrarDatos(data, qr) { document.getElementById('resultado').style.display = 'block'; // 1. Datos Generales (usando los campos del JSON real) document.getElementById('resQR').innerText = data.qr || qr; document.getElementById('resNombre').innerText = data.nombreCaja || 'Sin nombre'; const lat = data.latitud || 0; const lon = data.longitud || 0; document.getElementById('resCoords').innerText = `${lat}, ${lon}`; document.getElementById('resMaps').href = `https://www.google.com/maps/search/?api=1&query=${lat},${lon}`; // 2. Acceder al primer splitter y sus puertos const tabla = document.getElementById('tablaPuertos'); tabla.innerHTML = ""; // Verificamos que existan splitters if (data.splitters && data.splitters.length > 0) { const splitter = data.splitters[0]; // Tomamos el primer splitter const puertos = splitter.listmodelpuertos || []; // El JSON usa "1" para LIBRE, ajustamos el conteo const libres = puertos.filter(p => p.estatusPuerto == "1").length; document.getElementById('resLibres').innerHTML = `${libres} de ${puertos.length} disponibles`; puertos.forEach((p) => { // Estatus: 1 es Libre, cualquier otro suele ser Ocupado (Cuenta asignada) const esLibre = p.estatusPuerto == "1"; const estadoTexto = esLibre ? "LIBRE" : (p.cuenta || "OCUPADO"); const row = ` ${p.numeroPuerto} ${splitter.frame || 'N/A'} ${splitter.slot || 'N/A'} ${splitter.puerto || 'N/A'} ${p.numeroPuerto} ${estadoTexto} `; tabla.insertAdjacentHTML('beforeend', row); }); } else { tabla.innerHTML = "No se encontraron splitters en este QR"; } }
